cmake_minimum_required(VERSION 3.16)
project(optional_pkgconfig NONE)

function(disable_plugin name)
  set_property(GLOBAL PROPERTY disabled_provider "${name}")
endfunction()
function(enable_plugin name)
  message(FATAL_ERROR "Unexpected available provider: ${name}")
endfunction()

if(PROVIDER STREQUAL "pkcs11")
  # Exercise the pkg-config path of the helper finder on every host platform.
  set(WIN32 FALSE)
  list(PREPEND CMAKE_MODULE_PATH "${QCA_SOURCE_DIR}/cmake/modules")
  if(MODE STREQUAL "yes")
    find_package(Pkcs11Helper REQUIRED)
  else()
    find_package(Pkcs11Helper QUIET)
    if(PKCS11H_FOUND)
      message(FATAL_ERROR "Unexpected available PKCS11 helper")
    endif()
  endif()
else()
  set(WITH_${PROVIDER}_PLUGIN "${MODE}")
  add_subdirectory("${QCA_SOURCE_DIR}/plugins/qca-${PROVIDER}" provider)
  get_property(disabled GLOBAL PROPERTY disabled_provider)
  if(NOT disabled STREQUAL PROVIDER)
    message(FATAL_ERROR "Provider was not disabled: ${PROVIDER}")
  endif()
endif()
